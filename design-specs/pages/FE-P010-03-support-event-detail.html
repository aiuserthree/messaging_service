데이터 모델

interface EventDetail {
  id: string;
  title: string;
  content: string;
  thumbnailUrl: string;
  startDate: Date;
  endDate: Date;
  status: 'ongoing' | 'ended';
  eventType: EventType;
  participationInfo?: ParticipationInfo;
  createdAt: Date;
}

type EventType = 
  | 'charge_bonus'    // 충전 보너스
  | 'signup_bonus'    // 가입 혜택
  | 'referral'        // 추천인 이벤트
  | 'special';        // 특별 이벤트

interface ParticipationInfo {
  participated: boolean;
  participatedAt?: Date;
  details?: Record<string, any>;
}

비즈니스 로직

D-Day 계산 endDate - 현재일
상태 판단 - 현재일 < endDate: ongoing
- 현재일 >= endDate: ended
참여 가능 조건 - status === 'ongoing'
- !participationInfo.participated
