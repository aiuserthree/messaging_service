데이터 모델

interface SendResultDetail {
  sendId: string;
  sendDateTime: Date;
  messageType: 'SMS' | 'LMS' | 'MMS' | 'ALIMTALK' | 'BRANDTALK';
  sendType: 'GENERAL' | 'AD' | 'ELECTION';
  callerNumber: string;
  message: string;
  statistics: SendStatistics;
  recipients: RecipientResult[];
  pagination: Pagination;
}

interface SendStatistics {
  total: number;
  success: number;
  fail: number;
  cost: number;
}

interface RecipientResult {
  id: string;
  phoneNumber: string;  // 마스킹됨
  name?: string;
  status: 'SUCCESS' | 'FAILED' | 'PENDING';
  failReason?: FailReasonCode;
  sentAt: Date;
  cost?: number;
}

type FailReasonCode = 
  | 'INVALID_NUMBER'
  | 'REJECT_USER'
  | 'CARRIER_ERROR'
  | 'TIMEOUT'
  | 'UNKNOWN';
