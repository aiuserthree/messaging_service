발송 시간 검증

// 브랜드톡 발송 시간 제한 검증
const validateSendTime = (date: Date): boolean => {
  const hour = date.getHours();
  const day = date.getDay();
  
  // 주말(토:6, 일:0) 불가
  if (day === 0 || day === 6) return false;
  
  // 08:00~21:00만 허용
  if (hour < 8 || hour >= 21) return false;
  
  // 공휴일 체크 (별도 API)
  return !isHoliday(date);
};
